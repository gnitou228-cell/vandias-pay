3

Valider le Webhook instantané (HMAC-SHA256)

Dès que le paiement est validé, VandiasPay notifie votre serveur via un appel POST avec l'en-tête X-Vandias-Signature.

// Exemple de vérification Node.js / Express 
const crypto = require('crypto');

app.post('/webhooks/vandiaspay', express.raw({ type: 'application/json' }), (req, res) => {
  const signature = req.headers['x-vandias-signature'];
  const webhookSecret = process.env.VANDIAS_WEBHOOK_SECRET;

  const expectedSignature = crypto
    .createHmac('sha256', webhookSecret)
    .update(req.body)
    .digest('hex');

  if (signature !== expectedSignature) {
    return res.status(401).send('Signature invalide');
  }

  const event = JSON.parse(req.body.toString());
  if (event.event === 'payment.success') {
    const { amount, currency, metadata } = event.data;
    console.log(`Paiement reçu pour l'utilisateur ${metadata.user_id} !`);
    // Activer l'abonnement SaaS dans votre base de données...
  }

  res.json({ received: true });
});
